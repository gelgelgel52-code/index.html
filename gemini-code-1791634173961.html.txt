  <script>
    const songCatalog = {
      "001": { title: "Anak", artist: "Freddie Aguilar", ytId: "F3q3v3MJu0E" },
      "002": { title: "Harana", artist: "Parokya Ni Edgar", ytId: "5bsc_c6_Y6w" },
      "003": { title: "Torete", artist: "Moonstar88", ytId: "G9AC2EMxVXw" },
      "004": { title: "Narda", artist: "Kamikazee", ytId: "uJV0avAF_FU" },
      "005": { title: "Ang Huling El Bimbo", artist: "Eraserheads", ytId: "G9AC2EMxVXw" },
      "006": { title: "With a Smile", artist: "Eraserheads", ytId: "VwuOa2F9jk4" },
      "007": { title: "Halaga", artist: "Parokya Ni Edgar", ytId: "9iXre8k4x4c" },
      "008": { title: "Kisapmata", artist: "Rivermaya", ytId: "8di0DYO8kj0" }
    };

    let inputCode = "";
    let queue = [];
    let currentSong = null;

    const canvas = document.getElementById('videokeCanvas');
    const ctx = canvas.getContext('2d');

    function resizeCanvas() {
      canvas.width = canvas.clientWidth;
      canvas.height = canvas.clientHeight;
    }
    window.addEventListener('resize', resizeCanvas);
    resizeCanvas();

    function pressNum(num) {
      if (inputCode.length < 3) {
        inputCode += num;
        document.getElementById('codeDisplay').innerText = inputCode;
      }
    }

    function clearCode() {
      inputCode = "";
      document.getElementById('codeDisplay').innerText = currentSong ? currentSong.code : "---";
    }

    function enterCode() {
      if (songCatalog[inputCode]) {
        queue.push(inputCode);
        document.getElementById('queueDisplay').innerText = queue.length;
        if (!currentSong) playNextInQueue();
        inputCode = "";
      } else {
        alert("Song code not found! Choose between 001 - 008.");
        clearCode();
      }
    }

    function playNextInQueue() {
      if (queue.length > 0) {
        const nextCode = queue.shift();
        document.getElementById('queueDisplay').innerText = queue.length;
        currentSong = { ...songCatalog[nextCode], code: nextCode };
        document.getElementById('codeDisplay').innerText = nextCode;
        
        // I-load ang totoong YouTube Video Embed para tumugtog ang kanta at musika
        const playerContainer = document.getElementById('ytPlayerContainer');
        playerContainer.innerHTML = `<iframe src="https://www.youtube.com/embed/${currentSong.ytId}?autoplay=1&enablejsapi=1&rel=0" allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture" allowfullscreen></iframe>`;
      } else {
        currentSong = null;
        document.getElementById('codeDisplay').innerText = "---";
        document.getElementById('ytPlayerContainer').innerHTML = `<div style="display: flex; align-items: center; justify-content: center; height: 100%; color: #666; font-size: 0.9rem;">Wala pang naka-queue na kanta.</div>`;
      }
    }

    function nextSong() {
      playNextInQueue();
    }

    // Subtly background visual effect
    let step = 0;
    function renderScreen() {
      ctx.clearRect(0, 0, canvas.width, canvas.height);
      if (!currentSong) {
        step += 0.03;
        ctx.lineWidth = 2;
        for (let i = 0; i < 3; i++) {
          ctx.strokeStyle = `rgba(0, 245, 212, ${0.1 + i * 0.05})`;
          ctx.beginPath();
          for (let x = 0; x < canvas.width; x += 15) {
            let y = canvas.height / 2 + Math.sin(step + x * 0.02 + i) * 15;
            if (x === 0) ctx.moveTo(x, y); else ctx.lineTo(x, y);
          }
          ctx.stroke();
        }
      }
      requestAnimationFrame(renderScreen);
    }
    renderScreen();

    // Songbook Modal & Search
    function openSongbook() {
      renderSongList(songCatalog);
      document.getElementById('songbookModal').style.display = 'flex';
      document.getElementById('searchBox').value = '';
      document.getElementById('searchBox').focus();
    }

    function closeSongbook() {
      document.getElementById('songbookModal').style.display = 'none';
    }

    function renderSongList(catalog) {
      const listEl = document.getElementById('songList');
      listEl.innerHTML = "";
      Object.keys(catalog).forEach(code => {
        const item = document.createElement('div');
        item.className = 'song-item';
        item.innerHTML = `<div><strong>${catalog[code].title}</strong><br><small style="color:#aaa">${catalog[code].artist}</small></div><div class="song-code">${code}</div>`;
        item.onclick = () => { inputCode = code; enterCode(); closeSongbook(); };
        listEl.appendChild(item);
      });
    }

    function filterSongs() {
      const query = document.getElementById('searchBox').value.toLowerCase();
      const filtered = {};
      Object.keys(songCatalog).forEach(code => {
        const s = songCatalog[code];
        if (s.title.toLowerCase().includes(query) || s.artist.toLowerCase().includes(query) || code.includes(query)) {
          filtered[code] = s;
        }
      });
      renderSongList(filtered);
    }
