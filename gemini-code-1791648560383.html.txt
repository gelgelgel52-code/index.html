    let audioCtx = null;
    function playKeySound() {
      try {
        if (!audioCtx) {
          audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        }
        if (audioCtx.state === 'suspended') {
          audioCtx.resume();
        }
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        osc.type = 'sine';
        osc.frequency.setValueAtTime(800, audioCtx.currentTime);
        gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.1);
        osc.connect(gain);
        gain.connect(audioCtx.destination);
        osc.start();
        osc.stop(audioCtx.currentTime + 0.1);
      } catch (e) {}
    }

    function speakText(text) {
      if ('speechSynthesis' in window) {
        window.speechSynthesis.cancel();
        const utterance = new SpeechSynthesisUtterance(text);
        utterance.rate = 1.0;
        utterance.pitch = 0.9;
        utterance.lang = 'en-US';
        window.speechSynthesis.speak(utterance);
      }
    }

    function enterCode() {
      playKeySound();

        queue.push(inputCode);
        document.getElementById('queueDisplay').innerText = queue.length;
        speakText(`Reserved. ${selected.title}`);

        }
        inputCode = "";

    // --- DANCING BACKGROUND SIMULATION ---
    class Dancer {
      constructor(x, y, color) {
        this.x = x;
        this.y = y;
        this.color = color;
        this.size = Math.random() * 15 + 25;
        this.speed = Math.random() * 0.08 + 0.05;
        this.angle = Math.random() * Math.PI * 2;
      }
      update() {
        this.angle += this.speed;
      }
      draw(context) {
        context.save();
        context.strokeStyle = this.color;
        context.lineWidth = 3;
        context.fillStyle = this.color;

        let bounce = Math.sin(this.angle * 2) * 10;
        let cx = this.x;
        let cy = this.y + bounce;

        context.beginPath();
        context.arc(cx, cy - this.size, 8, 0, Math.PI * 2);
        context.fill();

        context.beginPath();
        context.moveTo(cx, cy - this.size + 8);
        context.lineTo(cx, cy + 10);
        context.stroke();

        let armAngle = Math.sin(this.angle) * 0.8;
        context.beginPath();
        context.moveTo(cx, cy - this.size + 15);
        context.lineTo(cx + Math.cos(armAngle) * 20, cy - this.size + 25 + Math.sin(armAngle) * 10);
        context.moveTo(cx, cy - this.size + 15);
        context.lineTo(cx - Math.cos(armAngle) * 20, cy - this.size + 25 - Math.sin(armAngle) * 10);
        context.stroke();

        let legAngle = Math.cos(this.angle) * 0.8;
        context.beginPath();
        context.moveTo(cx, cy + 10);
        context.lineTo(cx + 12 + legAngle * 10, cy + this.size);
        context.moveTo(cx, cy + 10);
        context.lineTo(cx - 12 - legAngle * 10, cy + this.size);
        context.stroke();

        context.restore();
      }
    }

    const dancers = [];
    for (let i = 0; i < 6; i++) {

    }

    let step = 0;
    function renderScreen() {
      ctx.clearRect(0, 0, canvas.width, canvas.height);
      
      step += 0.03;
      ctx.lineWidth = 2;
      for (let i = 0; i < 3; i++) {
        ctx.strokeStyle = `rgba(0, 245, 212, ${0.1 + i * 0.05})`;
        ctx.beginPath();
        for (let x = 0; x < canvas.width; x += 15) {
          let y = canvas.height / 2 + Math.sin(step + x * 0.02 + i) * 15;
          if (x === 0) ctx.moveTo(x, y); else ctx.lineTo(x, y);
        }
        ctx.stroke();
      }

      dancers.forEach(dancer => {
        dancer.update();
        dancer.draw(ctx);
      });

      requestAnimationFrame(renderScreen);
    }
    renderScreen();

      playKeySound();

      document.getElementById('songbookModal').style.display = 'flex';
      document.getElementById('searchBox').value = '';

    function closeSongbook() {
      document.getElementById('songbookModal').style.display = 'none';

    function renderSongList(codesArray) {
      const listEl = document.getElementById('songList');
      listEl.innerHTML = "";
      codesArray.forEach(code => {
        const s = songCatalog[code];
        const item = document.createElement('div');
        item.className = 'song-item';
        item.innerHTML = `<div><strong>${s.title}</strong><br><small style="color:#aaa">${s.artist}</small></div><div class="song-code">${code}</div>`;
        item.onclick = () => { inputCode = code; enterCode(); closeSongbook(); };
        listEl.appendChild(item);
      });
    }

    function filterSongs() {
      const query = document.getElementById('searchBox').value.toLowerCase();
      const matchedCodes = [];
      
      for (let code in songCatalog) {
        const s = songCatalog[code];
        if (s.title.toLowerCase().includes(query) || s.artist.toLowerCase().includes(query) || code.includes(query)) {
          matchedCodes.push(code);
        }
      }
      renderSongList(matchedCodes);
    }
