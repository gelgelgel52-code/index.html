  <script>
    // Audio Context & Speech Synthesis Setup
    let audioCtx = null;
    function playKeySound() {
      try {
        if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        if (audioCtx.state === 'suspended') audioCtx.resume();
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        osc.type = 'sine';
        osc.frequency.setValueAtTime(800, audioCtx.currentTime);
        gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.1);
        osc.connect(gain);
        gain.connect(audioCtx.destination);
        osc.start();
        osc.stop(audioCtx.currentTime + 0.1);
      } catch (e) {}
    }

    function speakText(text) {
      if ('speechSynthesis' in window) {
        window.speechSynthesis.cancel();
        const utterance = new SpeechSynthesisUtterance(text);
        utterance.rate = 1.0;
        utterance.pitch = 0.9;
        utterance.lang = 'en-US';
        window.speechSynthesis.speak(utterance);
      }
    }

    // Catalog Generation
    const songCatalog = {};
    const sampleArtists = ["Adie", "Eraserheads", "Parokya Ni Edgar", "Rivermaya", "Bamboo", "Side A", "December Avenue", "Moira Dela Torre", "Juan Karlos"];
    const sampleTitles = ["Paraluman", "Anak", "Harana", "Torete", "Narda", "Ang Huling El Bimbo", "With a Smile", "Halaga", "Kisapmata"];
    const fallbackYtIds = ["F3q3v3MJu0E", "5bsc_c6_Y6w", "G9AC2EMxVXw", "uJV0avAF_FU", "VwuOa2F9jk4", "9iXre8k4x4c", "8di0DYO8kj0"];

    for (let i = 1; i <= 2000; i++) {
      let code = i.toString().padStart(4, '0');
      let titleBase = sampleTitles[(i - 1) % sampleTitles.length];
      let artist = sampleArtists[(i - 1) % sampleArtists.length];
      let yt = fallbackYtIds[(i - 1) % fallbackYtIds.length];
      
      songCatalog[code] = {
        title: i <= 10 ? titleBase : `${titleBase} Vol. ${Math.ceil(i / sampleTitles.length)}`,
        artist: artist,
        ytId: yt
      };
    }

    let inputCode = "";
    let queue = [];
    let currentSong = null;

    const canvas = document.getElementById('videokeCanvas');
    const ctx = canvas.getContext('2d');

    function resizeCanvas() {
      canvas.width = canvas.clientWidth;
      canvas.height = canvas.clientHeight;
    }
    window.addEventListener('resize', resizeCanvas);
    resizeCanvas();

    function pressNum(num) {
      playKeySound();
      speakText(num);
      if (inputCode.length < 4) {
        inputCode += num;
        document.getElementById('codeDisplay').innerText = inputCode;
      }
    }

    function clearCode() {
      playKeySound();
      speakText("Clear");
      inputCode = "";
      document.getElementById('codeDisplay').innerText = currentSong ? currentSong.code : "----";
    }

    function enterCode() {
      playKeySound();
      if (songCatalog[inputCode]) {
        const selected = songCatalog[inputCode];
        queue.push(inputCode);
        document.getElementById('queueDisplay').innerText = queue.length;
        speakText(`Reserved. Song code ${inputCode}, ${selected.title}`);
        
        if (!currentSong) playNextInQueue();
        inputCode = "";
      } else {
        speakText("Song not found");
        alert("Song code not found! Pumili sa pagitan ng 0001 - 2000.");
        clearCode();
      }
    }

    function playNextInQueue() {
      if (queue.length > 0) {
        const nextCode = queue.shift();
        document.getElementById('queueDisplay').innerText = queue.length;
        currentSong = { ...songCatalog[nextCode], code: nextCode };
        document.getElementById('codeDisplay').innerText = nextCode;
        
        const playerContainer = document.getElementById('ytPlayerContainer');
        playerContainer.innerHTML = `<iframe src="https://www.youtube.com/embed/${currentSong.ytId}?autoplay=1&controls=0&modestbranding=1" allow="autoplay"></iframe>`;
      } else {
        currentSong = null;
        document.getElementById('codeDisplay').innerText = "----";
        document.getElementById('ytPlayerContainer').innerHTML = `<div style="display: flex; align-items: center; justify-content: center; height: 100%; color: #888; font-size: 0.9rem;">Wala pang naka-queue na kanta.</div>`;
      }
    }

    function nextSong() {
      playKeySound();
      speakText("Next song");
      playNextInQueue();
    }

    // --- ANIMATED DANCERS BACKGROUND ---
    class Dancer {
      constructor(x, y, color) {
        this.x = x;
        this.y = y;
        this.color = color;
        this.size = 35;
        this.speed = Math.random() * 0.1 + 0.08;
        this.angle = Math.random() * Math.PI * 2;
      }
      update() {
        this.angle += this.speed;
      }
      draw(context) {
        context.save();
        context.strokeStyle = this.color;
        context.lineWidth = 4;
        context.fillStyle = this.color;

        let bounce = Math.sin(this.angle * 2) * 15;
        let cx = this.x;
        let cy = this.y + bounce;

        // Ulo (Head)
        context.beginPath();
        context.arc(cx, cy - this.size, 10, 0, Math.PI * 2);
        context.fill();

        // Katawan (Body)
        context.beginPath();
        context.moveTo(cx, cy - this.size + 10);
        context.lineTo(cx, cy + 15);
        context.stroke();

        // Mga Braso (Dancing Arms)
        let armAngle = Math.sin(this.angle) * 1.2;
        context.beginPath();
        context.moveTo(cx, cy - this.size + 18);
        context.lineTo(cx + Math.cos(armAngle) * 25, cy - this.size + 30 + Math.sin(armAngle) * 15);
        context.moveTo(cx, cy - this.size + 18);
        context.lineTo(cx - Math.cos(armAngle) * 25, cy - this.size + 30 - Math.sin(armAngle) * 15);
        context.stroke();

        // Mga Paa (Dancing Legs)
        let legAngle = Math.cos(this.angle) * 1.0;
        context.beginPath();
        context.moveTo(cx, cy + 15);
        context.lineTo(cx + 15 + legAngle * 12, cy + this.size + 20);
        context.moveTo(cx, cy + 15);
        context.lineTo(cx - 15 - legAngle * 12, cy + this.size + 20);
        context.stroke();

        context.restore();
      }
    }

    const dancers = [
      new Dancer(80, 130, '#ff007f'),
      new Dancer(180, 140, '#00f5d4'),
      new Dancer(280, 130, '#fee440'),
      new Dancer(380, 140, '#7000ff')
    ];

    let discoAngle = 0;
    function renderScreen() {
      ctx.fillStyle = '#0a0a14';
      ctx.fillRect(0, 0, canvas.width, canvas.height);

      // Disco Lights Effect sa Background
      discoAngle += 0.02;
      let gradient = ctx.createRadialGradient(canvas.width / 2, canvas.height / 2, 10, canvas.width / 2, canvas.height / 2, 200);
      gradient.addColorStop(0, `hsla(${(discoAngle * 50) % 360}, 100%, 50%, 0.25)`);
      gradient.addColorStop(1, 'transparent');
      ctx.fillStyle = gradient;
      ctx.fillRect(0, 0, canvas.width, canvas.height);

      // I-render at iguhit ang mga sumasayaw na characters
      dancers.forEach(dancer => {
        dancer.update();
        dancer.draw(ctx);
      });

      requestAnimationFrame(renderScreen);
    }
    renderScreen();

    function openSongbook() {
      playKeySound();
      speakText("Songbook");
      renderSongList(Object.keys(songCatalog).slice(0, 100));
      document.getElementById('songbookModal').style.display = 'flex';
      document.getElementById('searchBox').value = '';
      document.getElementById('searchBox').focus();
    }

    function closeSongbook() {
      document.getElementById('songbookModal').style.display = 'none';
    }

    function renderSongList(codesArray) {
      const listEl = document.getElementById('songList');
      listEl.innerHTML = "";
      codesArray.forEach(code => {
        const s = songCatalog[code];
        const item = document.createElement('div');
        item.className = 'song-item';
        item.innerHTML = `<div><strong>${s.title}</strong><br><small style="color:#aaa">${s.artist}</small></div><div class="song-code">${code}</div>`;
        item.onclick = () => { inputCode = code; enterCode(); closeSongbook(); };
        listEl.appendChild(item);
      });
    }

    function filterSongs() {
      const query = document.getElementById('searchBox').value.toLowerCase();
      const matchedCodes = [];
      for (let code in songCatalog) {
        const s = songCatalog[code];
        if (s.title.toLowerCase().includes(query) || s.artist.toLowerCase().includes(query) || code.includes(query)) {
          matchedCodes.push(code);
          if (matchedCodes.length >= 150) break;
        }
      }
      renderSongList(matchedCodes);
    }
