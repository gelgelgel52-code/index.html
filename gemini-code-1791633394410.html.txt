  <script>
    const songCatalog = {
      "001": { title: "Anak", artist: "Freddie Aguilar", ytId: "F3q3v3MJu0E", lyrics: ["Nung isilang ka sa mundong ito", "Laking tuwa ng magulang mo", "At ang kamay nila ang iyong ilaw"] },
      "002": { title: "Harana", artist: "Parokya Ni Edgar", ytId: "5bsc_c6_Y6w", lyrics: ["Puno ang langit ng bituin", "At ang hangin ay malamig pa sa amin", "Pusong naghihintay"] },
      "003": { title: "Torete", artist: "Moonstar88", ytId: "G9AC2EMxVXw", lyrics: ["Sana'y 'di nalang umasa", "Sa 'yong mga pangako", "Torete sa 'yo"] },
      "004": { title: "Narda", artist: "Kamikazee", ytId: "uJV0avAF_FU", lyrics: ["Tumingin sa aking mga mata", "At sabihin nang tapat", "Tayo ay lilipad"] },
      "005": { title: "Ang Huling El Bimbo", artist: "Eraserheads", ytId: "G9AC2EMxVXw", lyrics: ["Kamukha mo si Paraluman", "Nung tayo ay bata pa", "At ang galing-galing mong sumayaw"] },
      "006": { title: "With a Smile", artist: "Eraserheads", ytId: "VwuOa2F9jk4", lyrics: ["Lift your head, baby, don't be afraid", "Though the road has been so dark", "And you can't see your way"] },
      "007": { title: "Halaga", artist: "Parokya Ni Edgar", ytId: "9iXre8k4x4c", lyrics: ["Nag-iisa ka man at 'di malaman", "Kung paano ang apektuhan", "Nasasaktan ka na pala"] },
      "008": { title: "Kisapmata", artist: "Rivermaya", ytId: "8di0DYO8kj0", lyrics: ["Nitong umaga lang", "Pagka ganda-ganda ng binuhat mo", "O, binata ka na"] }
    };

    let inputCode = "";
    let queue = [];
    let currentSong = null;
    let audioCtx = null;

    const canvas = document.getElementById('videokeCanvas');
    const ctx = canvas.getContext('2d');

    function resizeCanvas() {
      canvas.width = canvas.clientWidth;
      canvas.height = canvas.clientHeight;
    }
    window.addEventListener('resize', resizeCanvas);
    resizeCanvas();

    function initAudioContext() {
      if (!audioCtx) {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
      }
    }

    function playBeep(freq = 440, type = 'sine', duration = 0.1) {
      initAudioContext();
      if (!audioCtx) return;
      const osc = audioCtx.createOscillator();
      const gain = audioCtx.createGain();
      osc.type = type;
      osc.frequency.value = freq;
      gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
      osc.connect(gain);
      gain.connect(audioCtx.destination);
      osc.start();
      osc.stop(audioCtx.currentTime + duration);
    }

    function pressNum(num) {
      playBeep(600, 'square', 0.05);
      if (inputCode.length < 3) {
        inputCode += num;
        document.getElementById('codeDisplay').innerText = inputCode;
      }
    }

    function clearCode() {
      playBeep(300, 'sawtooth', 0.08);
      inputCode = "";
      document.getElementById('codeDisplay').innerText = currentSong ? currentSong.code : "---";
    }

    function enterCode() {
      if (songCatalog[inputCode]) {
        playBeep(800, 'sine', 0.15);
        queue.push(inputCode);
        document.getElementById('queueDisplay').innerText = queue.length;
        if (!currentSong) playNextInQueue();
        inputCode = "";
      } else {
        playBeep(150, 'sawtooth', 0.2);
        alert("Song code not found! Choose between 001 - 008.");
        clearCode();
      }
    }

    function playNextInQueue() {
      if (queue.length > 0) {
        const nextCode = queue.shift();
        document.getElementById('queueDisplay').innerText = queue.length;
        currentSong = { ...songCatalog[nextCode], code: nextCode };
        document.getElementById('codeDisplay').innerText = nextCode;
      } else {
        currentSong = null;
        document.getElementById('codeDisplay').innerText = "---";
      }
    }

    function nextSong() {
      playBeep(500, 'sine', 0.1);
      playNextInQueue();
    }

    // Animation Loop
    let step = 0;
    function renderScreen() {
      ctx.fillStyle = "#0d0221";
      ctx.fillRect(0, 0, canvas.width, canvas.height);

      step += 0.05;
      
      // Equalizer Wave FX
      ctx.lineWidth = 3;
      for (let i = 0; i < 5; i++) {
        ctx.strokeStyle = `hsl(${(step * 50 + i * 40) % 360}, 100%, 50%)`;
        ctx.beginPath();
        for (let x = 0; x < canvas.width; x += 10) {
          let y = canvas.height / 2 + Math.sin(step + x * 0.01 + i) * (20 + i * 8);
          if (x === 0) ctx.moveTo(x, y); else ctx.lineTo(x, y);
        }
        ctx.stroke();
      }

      if (currentSong) {
        ctx.fillStyle = "#ff007f";
        ctx.font = "bold 15px sans-serif";
        ctx.textAlign = "left";
        ctx.fillText(`NOW PLAYING [${currentSong.code}]: ${currentSong.title} - ${currentSong.artist}`, 20, 30);

        ctx.fillStyle = "#00f5d4";
        ctx.font = "bold 22px sans-serif";
        ctx.textAlign = "center";
        
        let lyrics = currentSong.lyrics;
        let line = lyrics[Math.floor(step / 3) % lyrics.length];
        ctx.fillText(line, canvas.width / 2, canvas.height / 2 + 50);
      } else {
        ctx.fillStyle = "#fee440";
        ctx.font = "bold 22px sans-serif";
        ctx.textAlign = "center";
        ctx.fillText("ENTER CODE (001 - 008) TO SING", canvas.width / 2, canvas.height / 2);
      }

      requestAnimationFrame(renderScreen);
    }
    renderScreen();

    // Songbook Modal & Search
    function openSongbook() {
      renderSongList(songCatalog);
      document.getElementById('songbookModal').style.display = 'flex';
      document.getElementById('searchBox').value = '';
      document.getElementById('searchBox').focus();
    }

    function closeSongbook() {
      document.getElementById('songbookModal').style.display = 'none';
    }

    function renderSongList(catalog) {
      const listEl = document.getElementById('songList');
      listEl.innerHTML = "";
      Object.keys(catalog).forEach(code => {
        const item = document.createElement('div');
        item.className = 'song-item';
        item.innerHTML = `<div><strong>${catalog[code].title}</strong><br><small style="color:#aaa">${catalog[code].artist}</small></div><div class="song-code">${code}</div>`;
        item.onclick = () => { inputCode = code; enterCode(); closeSongbook(); };
        listEl.appendChild(item);
      });
    }

    function filterSongs() {
      const query = document.getElementById('searchBox').value.toLowerCase();
      const filtered = {};
      Object.keys(songCatalog).forEach(code => {
        const s = songCatalog[code];
        if (s.title.toLowerCase().includes(query) || s.artist.toLowerCase().includes(query) || code.includes(query)) {
          filtered[code] = s;
        }
      });
      renderSongList(filtered);
    }
